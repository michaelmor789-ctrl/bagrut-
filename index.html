const parseBagrutText = (rawText) => { const text = rawText.replace(/[\r\n]+/g, ' ').replace(/\s+/g, ' '); const pattern = /(\d{1,2})[.:](\d{2})\s+(\d{2})\/(\d{2})\/(\d{4})\s+(.+?)\s+(\d{4,5})(?=\s|$)/g; const exams = []; let m; while ((m = pattern.exec(text)) !== null) { const hh = m[1], min = m[2], dd = m[3], mm = m[4], yyyy = m[5]; const chunk = m[6], shaalon = m[7]; const dN = parseInt(dd), mN = parseInt(mm), hN = parseInt(hh), minN = parseInt(min); if (dN < 1 || dN > 31 || mN < 1 || mN > 12 || hN > 23 || minN > 59) continue; let reversed = chunk.split('').reverse().join(''); reversed = reversed.replace(/\d{2,}/g, n => n.split('').reverse().join('')); const locMarkers = /(שלוחת|מכללת|מרכז\s|_?ברנר|אפעל|בית\s*ספר|אולפנת|ישיבת)/; const locIdx = reversed.search(locMarkers); let subject, note; if (locIdx > 0) { subject = reversed.substring(0, locIdx).trim(); note = reversed.substring(locIdx).trim(); } else { const parts = reversed.split(' '); subject = parts.slice(0, Math.min(3, parts.length)).join(' '); note = parts.slice(3).join(' ').trim(); } subject = subject.replace(/\s*-\s*תכנית\s*חדשה\s*/, ' ').trim(); if (!subject) subject = 'מקצוע'; const dtStr = `${yyyy}-${mm}-${dd}T${hh.padStart(2,'0')}:${min}:00`; exams.push({ subject: `${subject} ${shaalon}`, datetime: new Date(dtStr).toISOString(), note }); } const seen = new Set(); return exams.filter(e => { const k = `${e.datetime}|${e.subject}`; if (seen.has(k)) return false; seen.add(k); return true; }); };
